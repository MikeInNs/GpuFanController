cmake_minimum_required(VERSION 3.22)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/release.json" RELEASE_MANIFEST)
string(JSON RELEASE_VERSION GET "${RELEASE_MANIFEST}" version)
project(GpuFanController VERSION ${RELEASE_VERSION} LANGUAGES CXX)
add_compile_definitions(FAN_HOST_VERSION="${PROJECT_VERSION}")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(GNUInstallDirs)
include(CTest)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HTTPLIB REQUIRED IMPORTED_TARGET cpp-httplib)
find_package(nlohmann_json 3.11 REQUIRED)
include(cmake/Ftxui.cmake)

add_compile_options(-Wall -Wextra -Wpedantic -Werror)
add_library(fan_protocol src/protocol/Protocol.cpp)
target_include_directories(fan_protocol PUBLIC src/protocol/include)
add_library(fan_api_client src/api/ApiClient.cpp)
target_include_directories(fan_api_client PUBLIC src/api/include)
target_link_libraries(fan_api_client PUBLIC nlohmann_json::nlohmann_json PRIVATE PkgConfig::HTTPLIB)

add_library(fan_controller_data src/controller/ControllerData.cpp src/controller/AdapterNames.cpp)
target_include_directories(fan_controller_data PUBLIC src/controller)
target_link_libraries(fan_controller_data PUBLIC nlohmann_json::nlohmann_json)
add_library(fan_config src/daemon/ConfigStore.cpp)
target_include_directories(fan_config PUBLIC src/daemon)
target_link_libraries(fan_config PUBLIC nlohmann_json::nlohmann_json)
add_library(fan_hardware src/daemon/SerialProbe.cpp src/daemon/Discovery.cpp
  src/daemon/NvmlLibrary.cpp src/daemon/NvmlHelper.cpp src/daemon/NvmlReader.cpp
  src/daemon/GpuTemperatures.cpp src/daemon/AmdGpu.cpp src/daemon/GpuThermalLimits.cpp src/daemon/TemperatureForwarder.cpp src/daemon/TemperatureDiagnostics.cpp src/daemon/CalibrationControl.cpp
  src/daemon/AlertCenter.cpp src/daemon/AudibleAlarm.cpp src/daemon/GroupControl.cpp
  src/daemon/SettingsControl.cpp src/daemon/SerialSettings.cpp)
target_include_directories(fan_hardware PUBLIC src/daemon)
target_link_libraries(fan_hardware PUBLIC fan_config fan_protocol fan_controller_data nlohmann_json::nlohmann_json Threads::Threads ${CMAKE_DL_LIBS})
add_executable(gpu-fan-controllerd src/daemon/main.cpp src/daemon/ApiServer.cpp)
target_link_libraries(gpu-fan-controllerd PRIVATE fan_hardware PkgConfig::HTTPLIB)
add_library(fan_monitor src/monitor/CurveEditor.cpp src/monitor/MonitorModel.cpp src/monitor/StatusView.cpp
  src/monitor/LiveMonitor.cpp src/monitor/LiveView.cpp src/monitor/ConfigurationDraft.cpp src/monitor/SaveCoordinator.cpp src/monitor/ThermalSuggestion.cpp)
target_include_directories(fan_monitor PUBLIC src/monitor)
target_link_libraries(fan_monitor PUBLIC fan_config fan_api_client fan_controller_data ftxui::component ftxui::dom ftxui::screen Threads::Threads)
add_executable(fanctl src/monitor/main.cpp src/monitor/TerminalUi.cpp)
target_link_libraries(fanctl PRIVATE fan_api_client fan_monitor Threads::Threads)
install(TARGETS gpu-fan-controllerd fanctl RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
