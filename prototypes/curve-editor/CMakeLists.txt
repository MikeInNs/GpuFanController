cmake_minimum_required(VERSION 3.22)
project(FanCurvePrototype LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
include(FetchContent)
set(FTXUI_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(FTXUI_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(FTXUI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(FTXUI_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(ftxui
  GIT_REPOSITORY https://github.com/ArthurSonzogni/FTXUI.git
  GIT_TAG 5cfed50702f52d51c1b189b5f97f8beaf5eaa2a6 # v6.1.9
  GIT_PROGRESS TRUE)
FetchContent_MakeAvailable(ftxui)
add_library(curve_model CurveModel.cpp)
target_include_directories(curve_model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(curve_model PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(fanctl-curve-prototype main.cpp CurveEditor.cpp CurveGraph.cpp)
target_compile_options(fanctl-curve-prototype PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(fanctl-curve-prototype PRIVATE curve_model ftxui::component ftxui::dom ftxui::screen)
if(BUILD_TESTING)
  add_executable(curve_model_tests tests/model_tests.cpp)
  target_link_libraries(curve_model_tests PRIVATE curve_model)
  target_compile_options(curve_model_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME curve_model COMMAND curve_model_tests)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME curve_terminal COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/terminal_test.py $<TARGET_FILE:fanctl-curve-prototype>)
  set_tests_properties(curve_terminal PROPERTIES TIMEOUT 30)
endif()
